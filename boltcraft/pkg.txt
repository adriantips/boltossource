# The manifest. BoltOS's own `pkg build` reads this file, and so does the SDK
# on Windows -- `sources` is the one key the SDK added, and the kernel's parser
# skips keys it does not recognise, so this stays a valid pkg.txt either way.

package boltcraft
version 1.0
kind    program
summary a voxel sandbox -- dig, build, and watch the sun go down

# What to compile, relative to this directory. Add files as you add them.
sources main.c world.c tex.c render.c hud.c

# Where `pkg install` puts things, and what `pkg run boltcraft` starts.
install build/boltcraft /bin/boltcraft
run     /bin/boltcraft
